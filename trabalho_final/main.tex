% [CÓDIGO 1]: Exemplo básico de LLM usage
\begin{lstlisting}[caption={Exemplo de uso de LLM com LangChain},label={lst:llm-basic}]
from langchain.llms import OpenAI
from langchain.prompts import PromptTemplate
from langchain.chains import LLMChain

# Configurar LLM
llm = OpenAI(temperature=0.7, model="gpt-3.5-turbo")

# Criar prompt template
template = """Voce e um assistente especializado em {area}.
Pergunta: {pergunta}
Resposta:"""
prompt = PromptTemplate(
    input_variables=["area", "pergunta"],
    template=template
)

# Criar e executar chain
chain = LLMChain(llm=llm, prompt=prompt)
resposta = chain.run(
    area="inteligencia artificial",
    pergunta="O que sao agentes inteligentes?"
)
print(resposta)
\end{lstlisting}

\subsection{Retrieval e RAG}
\label{subsec:retrieval}

% [PARÁGRAFO 1 - 5-6 linhas]: Retrieval-Augmented Generation
% - Problema: LLMs têm conhecimento limitado ao training data
% - Solução: combinar LLM com busca em base de conhecimento externa
% - Processo: 1) Criar embeddings de documentos, 2) Buscar docs relevantes,
%   3) Incluir docs no contexto do LLM, 4) Gerar resposta fundamentada
% - Vantagens: informação atualizada, citações, redução de alucinações
% - Citar: \cite{lewis2020retrieval} para conceito de RAG

% [PARÁGRAFO 2 - 4-5 linhas]: Vector Stores
% - Armazenamento de embeddings para busca por similaridade
% - Integrações: Chroma, Pinecone, FAISS, Weaviate, Milvus
% - Operações: ingest (adicionar docs), search (buscar similares)
% - Métricas: cosine similarity, euclidean distance
% - Indexação para buscas eficientes

% [PARÁGRAFO 3 - 3-4 linhas]: Document Loaders
% - Carregamento de dados de múltiplas fontes
% - Suporte: PDF, Web, SQL, APIs, Google Drive, Notion
% - Text splitting: dividir documentos grandes em chunks
% - Metadata preservation: manter informações contextuais

% [CÓDIGO 2]: Exemplo de RAG implementation
\begin{lstlisting}[caption={Implementação básica de RAG com LangChain},label={lst:rag}]
from langchain.document_loaders import TextLoader
from langchain.text_splitter import CharacterTextSplitter
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI

# 1. Carregar documentos
loader = TextLoader("conhecimento.txt")
documents = loader.load()

# 2. Dividir em chunks
text_splitter = CharacterTextSplitter(
    chunk_size=1000,
    chunk_overlap=200
)
docs = text_splitter.split_documents(documents)

# 3. Criar embeddings e vector store
embeddings = OpenAIEmbeddings()
vectorstore = Chroma.from_documents(docs, embeddings)

# 4. Criar retriever
retriever = vectorstore.as_retriever(
    search_kwargs={"k": 3}
)

# 5. Criar QA chain
qa_chain = RetrievalQA.from_chain_type(
    llm=OpenAI(temperature=0),
    chain_type="stuff",
    retriever=retriever,
    return_source_documents=True
)

# 6. Fazer pergunta
resultado = qa_chain({"query": "Explique o conceito X"})
print(resultado["result"])
print("Fontes:", resultado["source_documents"])
\end{lstlisting}

\subsection{LangChain Expression Language}
\label{subsec:lcel}

% [PARÁGRAFO 1 - 4-5 linhas]: O que é LCEL
% - Sintaxe declarativa para compor chains
% - Operador pipe (|): conecta componentes
% - Inspirado em Unix pipes e functional programming
% - Vantagens: código mais legível, composição clara
% - Suporte nativo a streaming, async, parallelization

% [CÓDIGO 3]: Exemplo com LCEL
\begin{lstlisting}[caption={Exemplo de chain usando LCEL},label={lst:lcel}]
from langchain.prompts import ChatPromptTemplate
from langchain.chat_models import ChatOpenAI
from langchain.schema.output_parser import StrOutputParser

# Definir componentes
prompt = ChatPromptTemplate.from_template(
    "Traduza para {idioma}: {texto}"
)
model = ChatOpenAI()
parser = StrOutputParser()

# Compor chain usando LCEL
chain = prompt | model | parser

# Executar
resultado = chain.invoke({
    "idioma": "frances",
    "texto": "Hello, how are you?"
})
print(resultado)  # "Bonjour, comment allez-vous?"
\end{lstlisting}
